    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    
    function playSound() {
        if (audioCtx.state === 'suspended') { audioCtx.resume(); }
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(587.33, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.1);
    }

    function playCorrectSound() {
        if (audioCtx.state === 'suspended') { audioCtx.resume(); }
        const now = audioCtx.currentTime;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(523.25, now);
        osc.frequency.setValueAtTime(659.25, now + 0.1);
        osc.frequency.setValueAtTime(783.99, now + 0.2);
        gain.gain.setValueAtTime(0.15, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start(now);
        osc.stop(now + 0.4);
    }

    function playWrongSound() {
        if (audioCtx.state === 'suspended') { audioCtx.resume(); }
        const now = audioCtx.currentTime;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(220, now);
        osc.frequency.setValueAtTime(164.81, now + 0.15);
        gain.gain.setValueAtTime(0.2, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start(now);
        osc.stop(now + 0.4);
    }

    function checkLiveState() {
        const data = levelsData[currentLevel - 1];
        
        if(data.gridType === 1) {
            let s1 = grid[0] + grid[1] + grid[2];
            let s2 = grid[3] + grid[4];
            let s3 = grid[5] + grid[6];
            let s4 = grid[7] + grid[8];

            toggleSolved([0,1,2], s1 === 7);
            toggleSolved([3,4], s2 === 5);
            toggleSolved([5,6], s3 === 6);
            toggleSolved([7,8], s4 === 5);
        } else if(data.gridType === 2) {
            let s1 = grid[0] + grid[1] + grid[2] + grid[3];
            let s2 = grid[4] + grid[5];
            let s3 = grid[6] + grid[7];
            let s4 = grid[8] + grid[9];

            toggleSolved([0,1,2,3], s1 === 10);
            toggleSolved([4,5], s2 === 5);
            toggleSolved([6,7], s3 === 6);
            toggleSolved([8,9], s4 === 5);
        } else if(data.gridType === 3) {
            let s1 = grid[0] + grid[1] + grid[2] + grid[3];
            let s2 = grid[4] + grid[5] + grid[6];
            let s3 = grid[7] + grid[8];
            let s4 = grid[9];

            toggleSolved([0,1,2,3], s1 === 11);
            toggleSolved([4,5,6], s2 === 8);
            toggleSolved([7,8], s3 === 5);
            toggleSolved([9], s4 === 3);
        }
    }

    function toggleSolved(indices, condition) {
        indices.forEach(idx => {
            const el = document.getElementById(`c${idx}`);
            if(el) {
                if(condition) el.classList.add("solved");
                else el.classList.remove("solved");
            }
        });
    }

    function checkPuzzle() {

            if((grid[0]+grid[1]+grid[2]) === 7 && (grid[3]+grid[4]) === 5 && (grid[5]+grid[6]) === 6 && (grid[7]+grid[8]) === 5) isValid = true;

        }

            playCorrectSound();
            container.classList.add("flash-correct");

    }

    function gameOver(reason) {
        document.getElementById("screen-game").classList.add("hidden");
        document.getElementById("screen-question").classList.add("hidden");
        document.getElementById("failReason").innerText = reason;
        document.getElementById("screen-gameover").classList.remove("hidden");
    }

    function loadQuestion() {
        const qData = levelsData[currentLevel - 1].question;
        document.getElementById("questionText").innerText = qData.q;
        const optContainer = document.getElementById("optionsContainer");
        optContainer.innerHTML = "";
        qData.options.forEach((opt, idx) => {
            const btn = document.createElement("button");
            btn.className = "option-btn";
            btn.innerText = opt;
            btn.onclick = () => { checkAnswer(idx, qData.answer); };
            optContainer.appendChild(btn);
        });
    }

    function checkAnswer(selected, correct) {
        const container = document.getElementById("app");
        if(selected === correct) {
            score += 5;
            playCorrectSound();
            container.classList.add("flash-correct");
            setTimeout(() => { container.classList.remove("flash-correct"); }, 1000);
        } else {
            playWrongSound();
            container.classList.add("flash-wrong");
            setTimeout(() => { container.classList.remove("flash-wrong"); }, 1000);
        }
        setTimeout(() => { nextLevelOrFinish(); }, 1000);
    }

    function nextLevelOrFinish() {
        currentLevel++;
        if(currentLevel <= 3) {
            document.getElementById("screen-question").classList.add("hidden");
            document.getElementById("screen-game").classList.remove("hidden");
            loadLevel();
        } else {
            showLeaderboard();
        }
    }

    // Yerel Hafıza Destekli Anında Yüklenen Liderlik Tablosu
    function showLeaderboard() {
        document.getElementById("screen-question").classList.add("hidden");
        document.getElementById("screen-leaderboard").classList.remove("hidden");
        
        let lb = JSON.parse(localStorage.getItem("tubitak_lb_v24") || "[]");
        lb.push({ name: playerName, score: score });
        lb.sort((a,b) => b.score - a.score);
        localStorage.setItem("tubitak_lb_v24", JSON.stringify(lb));

        const tbody = document.getElementById("leaderboardBody");
        tbody.innerHTML = "";
        
        lb.forEach((item, index) => {
            let medal = "";
            if(index === 0) medal = "🥇 ";
            else if(index === 1) medal = "🥈 ";
            else if(index === 2) medal = "🥉 ";

            tbody.innerHTML += `
                <tr>
                    <td><strong>${medal}${index + 1}.</strong></td>
                    <td>${item.name}</td>
                    <td>${item.score} / 15 Puan</td>
                </tr>
            `;
        });
    }
